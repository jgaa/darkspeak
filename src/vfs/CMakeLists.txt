cmake_minimum_required(VERSION 3.24)
project(DarkSpeakVfs LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Sql)
if(NOT TARGET DSFileManager::FileManager)
    add_subdirectory(../filemanager ${CMAKE_CURRENT_BINARY_DIR}/filemanager)
endif()
if(NOT TARGET logfault::logfault)
    find_package(logfault REQUIRED CONFIG)
    if(NOT TARGET logfault::logfault AND EXISTS "${logfault_DIR}/logfaultTargets.cmake")
        include("${logfault_DIR}/logfaultTargets.cmake")
    endif()
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(VFS_SODIUM REQUIRED IMPORTED_TARGET libsodium)
find_package(SQLite3 REQUIRED)
add_library(dsvfs src/vfs.cpp src/policy.cpp include/dsvfs/policy.h include/dsvfs/vfs.h)
target_compile_features(dsvfs PUBLIC cxx_std_20)
target_include_directories(dsvfs PUBLIC include)
target_link_libraries(dsvfs PUBLIC DSFileManager::FileManager Qt6::Core PRIVATE PkgConfig::VFS_SODIUM SQLite::SQLite3 logfault::logfault)
target_compile_options(dsvfs PRIVATE -Wall -Wextra -Wpedantic)
if(DSFM_BUILD_TESTS)
    enable_testing()
    add_executable(vfs_test tests/vfs_test.cpp)
    target_link_libraries(vfs_test PRIVATE dsvfs)
    target_compile_definitions(vfs_test PRIVATE VFS_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    add_test(NAME vfs COMMAND vfs_test)
    set_tests_properties(vfs PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
endif()
