cmake_minimum_required(VERSION 3.24)
project(DSFileManager VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(FetchContent)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The native backend currently requires Linux directory-handle and SQLite VFS helpers")
endif()
option(DSFM_BUILD_TESTS "Build standalone file manager tests" ON)
option(DSFM_BUILD_EXAMPLE "Build the standalone example" ON)
set(DSFM_QCORO_SOURCE_DIR "" CACHE PATH "Offline QCoro source checkout")
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Sql)
find_package(QCoro6Core QUIET)
if(NOT TARGET QCoro6::Core)
    set(QCORO_BUILD_TESTING OFF CACHE BOOL "" FORCE)
    set(QCORO_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    foreach(component QTDBUS QTNETWORK QTWEBSOCKETS QTQUICK QML QTTEST)
        set(QCORO_WITH_${component} OFF CACHE BOOL "" FORCE)
    endforeach()
    if(DSFM_QCORO_SOURCE_DIR)
        FetchContent_Declare(qcoro SOURCE_DIR "${DSFM_QCORO_SOURCE_DIR}")
    else()
        FetchContent_Declare(qcoro GIT_REPOSITORY https://github.com/danvratil/qcoro.git GIT_TAG main GIT_SHALLOW TRUE)
    endif()
    FetchContent_MakeAvailable(qcoro)
endif()
find_package(logfault QUIET CONFIG)
if(logfault_FOUND AND NOT TARGET logfault::logfault AND EXISTS "${logfault_DIR}/logfaultTargets.cmake")
    include("${logfault_DIR}/logfaultTargets.cmake")
endif()
if(NOT TARGET logfault::logfault)
    set(DSFM_LOGFAULT_SOURCE_DIR "" CACHE PATH "Offline logfault source checkout")
    if(DSFM_LOGFAULT_SOURCE_DIR)
        FetchContent_Declare(logfault SOURCE_DIR "${DSFM_LOGFAULT_SOURCE_DIR}")
    elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../dependencies/logfault/CMakeLists.txt")
        FetchContent_Declare(logfault SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../dependencies/logfault")
    else()
        FetchContent_Declare(logfault GIT_REPOSITORY https://github.com/jgaa/logfault.git GIT_TAG master GIT_SHALLOW TRUE)
    endif()
    FetchContent_MakeAvailable(logfault)
endif()
find_package(Threads REQUIRED)
find_package(SQLite3 REQUIRED)
add_library(dsfilemanager src/filemanager.cpp src/native.cpp src/catalog.cpp src/names.cpp src/recovery.cpp src/sqlitevfs.cpp)
add_library(DSFileManager::FileManager ALIAS dsfilemanager)
set_target_properties(dsfilemanager PROPERTIES EXPORT_NAME FileManager)
target_compile_features(dsfilemanager PUBLIC cxx_std_20)
target_compile_definitions(dsfilemanager PRIVATE _FILE_OFFSET_BITS=64)
target_include_directories(dsfilemanager PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(dsfilemanager PUBLIC Qt6::Core QCoro6::Core PRIVATE Qt6::Sql logfault::logfault Threads::Threads SQLite::SQLite3)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(dsfilemanager PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS dsfilemanager EXPORT DSFileManagerTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES README.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(FILES LICENSE.unicode.txt DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/licenses/DSFileManager)
install(EXPORT DSFileManagerTargets NAMESPACE DSFileManager:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/DSFileManager)
configure_package_config_file(cmake/DSFileManagerConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/DSFileManagerConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/DSFileManager)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/DSFileManagerConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/DSFileManagerConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/DSFileManagerConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/DSFileManager)
if(DSFM_BUILD_TESTS)
    enable_testing()
    find_package(Qt6 REQUIRED COMPONENTS Test)
    get_target_property(DSFM_SOURCES dsfilemanager SOURCES)
    add_library(dsfm_test_support STATIC ${DSFM_SOURCES})
    target_compile_features(dsfm_test_support PUBLIC cxx_std_20)
    target_include_directories(dsfm_test_support PUBLIC include)
    target_compile_definitions(dsfm_test_support PRIVATE DSFM_FAULT_INJECTION _FILE_OFFSET_BITS=64)
    target_link_libraries(dsfm_test_support PUBLIC Qt6::Core QCoro6::Core PRIVATE Qt6::Sql SQLite::SQLite3 logfault::logfault Threads::Threads)
    add_executable(dsfm_tests tests/filemanager_test.cpp)
    target_link_libraries(dsfm_tests PRIVATE dsfm_test_support Qt6::Test Qt6::Sql logfault::logfault)
    target_compile_definitions(dsfm_tests PRIVATE DSFM_TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests")
    add_test(NAME filemanager COMMAND dsfm_tests)
    set_tests_properties(filemanager PROPERTIES TIMEOUT 120)
endif()
if(DSFM_BUILD_EXAMPLE)
    add_executable(dsfm_example examples/stream.cpp)
    target_link_libraries(dsfm_example PRIVATE dsfilemanager)
endif()
